    i = 2

    while n > 1:
        if n % i == 0:
            print(i, end=" ")
            n = n // i
        else:
            i = i + 1

3)import math

try:
    a = float(input("Enter value of a: "))
    b = float(input("Enter value of b: "))
    c = float(input("Enter value of c: "))

    d = b * b - 4 * a * c

    if d >= 0:
        root1 = (-b + math.sqrt(d)) / (2 * a)
        root2 = (-b - math.sqrt(d)) / (2 * a)

        print("Root 1 =", root1)
        print("Root 2 =", root2)
    else:
        print("math domain error")

except ValueError:
