marks = int(input("Enter marks: "))

if marks >= 90:

    print("Grade: O")

elif marks >= 80:

    print("Grade: A+")

elif marks >= 70:

    print("Grade: A")

elif marks >= 60:

    print("Grade: B")

elif marks >= 50:

    print("Grade: C")

elif marks >= 40:

    print("Grade: P")

else:

    print("Grade: F")




Output:
Enter marks: 85
Grade: A+
