7)string = input("Enter a string: ")

frequency = {}

for letter in string:
    if letter in frequency:
        frequency[letter] = frequency[letter] + 1
    else:
        frequency[letter] = 1

print("Frequency of letters:")
print(frequency)

Output:Enter a string: hello
Frequency of letters:
{'h': 1, 'e': 1, 'l': 2, 'o': 1}
